printf '%s\n' '--- files ---'
find . -mindepth 1 -maxdepth 1 -printf '%f\t%s bytes\t%M\n'

printf '%s\n' '--- tracked files ---'
git ls-files

printf '%s\n' '--- auxiliary files (redacted scan) ---'
for f in settingKeycloak.txt tmpError.txt; do
  if [[ -f "$f" ]]; then
    stat -c '%n: %s bytes' "$f"
    grep -Ein '(password|secret|token|credential|error|exception|failed)' "$f" |
      sed -E 's/(password|secret|token|credential)([[:space:]]*[:=][[:space:]]*)[^[:space:]]+/\1\2[REDACTED]/Ig'
  fi
done
