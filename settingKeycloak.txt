apiVersion: v1
kind: Pod
metadata:
  name: pg-mesh-probe
  namespace: progetto
  annotations:
    linkerd.io/inject: enabled
spec:
  restartPolicy: Never
  containers:
    - name: client
      image: docker.io/bitnamilegacy/postgresql:17.6.0-debian-12-r0
      command: ["/bin/bash", "-ec"]
      args:
        - |
          export PGPASSWORD="$(cat /db/password)"
          export PGSSLMODE=disable
          psql -h mio-keycloak-postgresql -p 5432 \
            -U admin -d keycloak_db \
            -Atc 'select current_user, current_database();'
      volumeMounts:
        - name: db-secret
          mountPath: /db
          readOnly: true
  volumes:
    - name: db-secret
      secret:
        secretName: mio-keycloak-postgresql
        items:
          - key: password
            path: password


minikube kubectl -- apply -f pg-mesh-probe.yaml
minikube kubectl -- get pod pg-mesh-probe -n progetto
minikube kubectl -- logs pg-mesh-probe -n progetto -c client


minikube kubectl -- delete pod pg-mesh-probe -n progetto
