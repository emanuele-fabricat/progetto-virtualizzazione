NS="namespace"
POD="nome-pod"

minikube kubectl -- get pod "$POD" -n "$NS" \
  -o jsonpath='{range .spec.volumes[*]}{.name}{"\t"}{.persistentVolumeClaim.claimName}{"\n"}{end}'

minikube kubectl -- get pod "$POD" -n "$NS" \
  -o jsonpath='{.metadata.ownerReferences[0].kind}{"/"}{.metadata.ownerReferences[0].name}{"\n"}'
